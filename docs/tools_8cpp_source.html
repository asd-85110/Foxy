23namespace Tools {
24GLFWimage*
25load_png(const char* filename, int* width, int* height) {
26 FILE* fp = fopen(filename, "rb");
27 if(!fp)
28 return nullptr;
29 png_byte header[8];
30 fread(header, 1, 8, fp);
31 if(png_sig_cmp(header, 0, 8)) {
32 fclose(fp);
33 return nullptr;
34 }
35 png_structp png = png_create_read_struct(PNG_LIBPNG_VER_STRING, nullptr, nullptr, nullptr);
36 if(!png) {
37 fclose(fp);
38 return nullptr;
39 }
40 png_infop info = png_create_info_struct(png);
41 if(!info) {
42 png_destroy_read_struct(&png, nullptr, nullptr);
43 fclose(fp);
44 return nullptr;
45 }
46 if(setjmp(png_jmpbuf(png))) {
47 png_destroy_read_struct(&png, &info, nullptr);
48 fclose(fp);
49 return nullptr;
50 }
51 png_init_io(png, fp);
52 png_set_sig_bytes(png, 8);
53 png_read_info(png, info);
54 int w = png_get_image_width(png, info);
55 int h = png_get_image_height(png, info);
56 png_byte color_type = png_get_color_type(png, info);
57 png_byte bit_depth = png_get_bit_depth(png, info);
58 if(bit_depth == 16) {
59 png_set_strip_16(png);
60 }
61 if(color_type == PNG_COLOR_TYPE_PALETTE) {
62 png_set_palette_to_rgb(png);
63 }
64 if(color_type == PNG_COLOR_TYPE_GRAY && bit_depth < 8) {
65 png_set_expand_gray_1_2_4_to_8(png);
66 }
67 if(png_get_valid(png, info, PNG_INFO_tRNS))
68 png_set_tRNS_to_alpha(png);
69 if(color_type == PNG_COLOR_TYPE_RGB || color_type == PNG_COLOR_TYPE_GRAY || color_type == PNG_COLOR_TYPE_PALETTE)
70 png_set_filler(png, 0xFF, PNG_FILLER_AFTER);
71 if(color_type == PNG_COLOR_TYPE_GRAY || color_type == PNG_COLOR_TYPE_GRAY_ALPHA)
72 png_set_gray_to_rgb(png);
73 png_read_update_info(png, info);
74 try {
75 png_bytep* row_pointers = new png_bytep[h];
76 for(int y = 0; y < h; ++y) {
77 row_pointers[y] = new png_byte[png_get_rowbytes(png, info)];
78 }
79 png_read_image(png, row_pointers);
80 fclose(fp);
81 GLFWimage* image = new GLFWimage;
82 image->width = w;
83 image->height = h;
84 image->pixels = new unsigned char[w * h * 4];
85 for(int y = 0; y < h; ++y) {
86 memcpy(image->pixels + y * w * 4, row_pointers[y], w * 4);
87 }
88 for(int y = 0; y < h; ++y) {
89 delete[] row_pointers[y];
90 }
91 delete[] row_pointers;
92 png_destroy_read_struct(&png, &info, nullptr);
93 if(width)
94 *width = w;
95 if(height)
96 *height = h;
97 return image;
98 } catch(const std::bad_alloc&) {
99 png_destroy_read_struct(&png, &info, nullptr);
100 fclose(fp);
101 return nullptr;
102 }
103}
